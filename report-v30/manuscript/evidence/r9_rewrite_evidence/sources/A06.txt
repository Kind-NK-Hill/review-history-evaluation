def _all_sessions_distinct(results: Iterable[dict[str, Any]]) -> bool:

def advance_endpoint(case_dir: Path) -> dict[str, Any]:
    """Derive the next state; write at most one adjudication request or decision."""
    case = Path(case_dir).resolve()
    final = case / "final/decision.json"
    if final.exists():
        decision = _read_object(final)
        return {"status": "complete", "verdict": decision["verdict"], "decision": str(final)}
    review_paths = [case / "results/reviewer-1.json", case / "results/reviewer-2.json"]
    if not all(path.exists() for path in review_paths):
        return {"status": "awaiting_reviewers", "received": sum(path.exists() for path in review_paths), "required": 2}
    reviews = [_read_object(path) for path in review_paths]
    requests = [_read_object(case / f"requests/reviewer-{slot}.json") for slot in (1, 2)]
    for result, request in zip(reviews, requests):
        _validate_evaluation(result, request, "reviewer")
    if not _all_sessions_distinct(reviews):
        raise EndpointContractError("the two endpoint reviewers are not independent")
    verdicts = [row["verdict"] for row in reviews]
    if verdicts[0] == verdicts[1]:
        verdict = verdicts[0] if verdicts[0] in {"pass", "fail"} else "unresolved"
        return _finish(case, verdict, reviews, None)
    arbitration = case / "arbitration/request.json"
    if not arbitration.exists():
        _write_json_new(arbitration, _arbitration_request(case, reviews[0], reviews[1]))
        _write_json_new(case / "private/arbitration-execution-job.json", {
            "schema": "formal_endpoint.offline_execution_job.v1",
            "evidence_namespace": _read_object(case / "input-binding.json")["evidence_namespace"],
            "slot": "adjudicator",
            "subject_host_path": str((case / "subject").resolve()),
            "subject_container_path": "/endpoint/input",
            "subject_read_only": True,
            "request_path": str(arbitration.resolve()),
            "prompt_field": "prompt",
            "model": MODEL,
            "reasoning_effort": EFFORT,
            "fresh_session": True,
            "result_record_command": ["record", str(case), "adjudicator", "<result-json>"],
            "model_calls_started": 0,
        })
    adjudication_result = case / "results/adjudicator.json"
    if not adjudication_result.exists():
        return {"status": "awaiting_adjudication", "request": str(arbitration), "requests_created": 1}
    adjudicator = _read_object(adjudication_result)
    _validate_evaluation(adjudicator, _read_object(arbitration), "adjudicator")
    verdict = adjudicator["verdict"] if adjudicator["verdict"] in {"pass", "fail"} else "unresolved"
    return _finish(case, verdict, reviews, adjudicator)
